CODE EXAMPLE (NEW):
import requests

class RiskAssessmentAgent:
    """Uses AIML API for risk scoring"""
    
    def assess_payment_risk(self, contract_data, user_history):
        url = "https://api.aimlapi.com/v1/chat/completions"
        
        response = requests.post(url,
            headers={
                "Authorization": f"Bearer {AIML_API_KEY}",
                "Content-Type": "application/json"
            },
            json={
                "model": "gpt-3.5-turbo",  # Or other AIML models
                "messages": [
                    {"role": "system", "content": "You are a risk assessment expert"},
                    {"role": "user", "content": f"Assess risk: {contract_data}"}
                ]
            }
        )
        
        return response.json()
